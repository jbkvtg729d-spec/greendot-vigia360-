<script>
const canvas = document.getElementById('mapCanvas');
const ctx = canvas.getContext('2d');

let modoPrivado = false; 
let ecosistemaActual = 'ubiquiti';
let elementos = [];

function resizeCanvas() {
  canvas.width = canvas.parentElement.clientWidth;
  canvas.height = canvas.parentElement.clientHeight;
  dibujarMapa();
}
window.addEventListener('resize', resizeCanvas);

function dibujarMapa() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  
  ctx.fillStyle = '#0b0f15';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  ctx.strokeStyle = 'rgba(255,255,255,0.03)';
  ctx.lineWidth = 1;
  for(let x=0; x<canvas.width; x+=50) {
    ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, canvas.height); ctx.stroke();
  }
  for(let y=0; y<canvas.height; y+=50) {
    ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(canvas.width, y); ctx.stroke();
  }

  // Polígono base del terreno
  ctx.strokeStyle = '#10b981';
  ctx.lineWidth = 3;
  ctx.beginPath();
  ctx.moveTo(canvas.width * 0.3, canvas.height * 0.2);
  ctx.lineTo(canvas.width * 0.7, canvas.height * 0.25);
  ctx.lineTo(canvas.width * 0.8, canvas.height * 0.65);
  ctx.lineTo(canvas.width * 0.4, canvas.height * 0.8);
  ctx.lineTo(canvas.width * 0.25, canvas.height * 0.45);
  ctx.closePath();
  ctx.stroke();
  ctx.fillStyle = 'rgba(16, 185, 129, 0.05)';
  ctx.fill();

  elementos.forEach(el => {
    ctx.save();
    ctx.translate(el.x, el.y);
    
    if(el.tipo === 'camara') {
      ctx.fillStyle = 'rgba(16, 185, 129, 0.25)';
      ctx.beginPath();
      ctx.moveTo(0,0);
      ctx.arc(0, 0, 60, -0.5, 0.5);
      ctx.closePath();
      ctx.fill();

      ctx.beginPath(); ctx.arc(0, 0, 10, 0, Math.PI*2);
      ctx.fillStyle = '#10b981'; ctx.fill();
      ctx.strokeStyle = '#fff'; ctx.lineWidth = 2; ctx.stroke();
    } else if(el.tipo === 'nodo') {
      ctx.beginPath();
      ctx.moveTo(0, -12); ctx.lineTo(-10, 8); ctx.lineTo(10, 8); ctx.closePath();
      ctx.fillStyle = '#0284c7'; ctx.fill();
      ctx.strokeStyle = '#fff'; ctx.lineWidth = 2; ctx.stroke();
    } else if(el.tipo === 'sirena') {
      ctx.beginPath(); ctx.arc(0, 0, 10, 0, Math.PI*2);
      ctx.fillStyle = '#dc2626'; ctx.fill();
      ctx.strokeStyle = '#fff'; ctx.lineWidth = 2; ctx.stroke();
    }

    ctx.restore();
  });
}

function cambiarEcosistema() {
  ecosistemaActual = document.getElementById('ecoSelect').value;
  actualizarPanel();
}

function toggleModo() {
  modoPrivado = !modoPrivado;
  const badge = document.getElementById('modeBadge');
  if(modoPrivado) {
    badge.className = 'mode-badge mode-private';
    badge.textContent = 'Modo Privado (Ingeniería)';
  } else {
    badge.className = 'mode-badge mode-client';
    badge.textContent = 'Modo Cliente';
  }
  actualizarPanel();
}

function ejecutarIA() {
  elementos = [
    { tipo: 'nodo', x: canvas.width * 0.5, y: canvas.height * 0.22, nombre: 'Nodo Central 5GHz' },
    { tipo: 'camara', x: canvas.width * 0.35, y: canvas.height * 0.3, nombre: 'Cámara Perimetral 1' },
    { tipo: 'camara', x: canvas.width * 0.65, y: canvas.height * 0.32, nombre: 'Cámara Perimetral 2' },
    { tipo: 'camara', x: canvas.width * 0.75, y: canvas.height * 0.55, nombre: 'Cámara Perimetral 3' },
    { tipo: 'camara', x: canvas.width * 0.5, y: canvas.height * 0.75, nombre: 'Cámara Perimetral 4' },
    { tipo: 'sirena', x: canvas.width * 0.77, y: canvas.height * 0.6, nombre: 'Sirena Zona Este' }
  ];
  dibujarMapa();
  actualizarPanel();
}

function agregarElemento(tipo) {
  elementos.push({
    tipo: tipo,
    x: canvas.width / 2 + (Math.random() * 60 - 30),
    y: canvas.height / 2 + (Math.random() * 60 - 30),
    nombre: tipo.toUpperCase() + ' ' + (elementos.length + 1)
  });
  dibujarMapa();
  actualizarPanel();
}

function actualizarPanel() {
  document.getElementById('itemCount').textContent = elementos.length + ' elementos';
  
  let listaHTML = '';
  let costoTotal = 0;

  elementos.forEach((el, idx) => {
    let nombreMostrar = el.nombre;
    let precioMostrar = '';

    if(ecosistemaActual === 'ubiquiti') {
      if(!modoPrivado) {
        nombreMostrar = el.tipo === 'camara' ? 'Cámara Bala 2K IA Exterior' : (el.tipo === 'nodo' ? 'Antena Troncal 5 GHz' : 'Sirena de Alerta 120dB');
      } else {
        nombreMostrar = el.tipo === 'camara' ? 'UniFi Protect G5 Bullet' : (el.tipo === 'nodo' ? 'LiteAP GPS 5GHz' : 'Sirena IP Relé');
        precioMostrar = el.tipo === 'camara' ? '$140.00' : (el.tipo === 'nodo' ? '$120.00' : '$85.00');
        costoTotal += el.tipo === 'camara' ? 140 : (el.tipo === 'nodo' ? 120 : 85);
      }
    } else {
      if(!modoPrivado) {
        nombreMostrar = el.tipo === 'camara' ? 'Cámara Wi-Fi HD Externa' : 'Dispositivo de Red';
      } else {
        nombreMostrar = el.tipo === 'camara' ? 'Tapo C120 / C325WB' : 'Router Repetidor';
        precioMostrar = '$55.00';
        costoTotal += 55;
      }
    }

    listaHTML += `<div style="display:flex; justify-content:space-between; font-size:12px; border-bottom:1px solid #222; padding-bottom:4px;">
      <span>${nombreMostrar}</span>
      <span style="color:var(--accent-green); font-weight:700;">${modoPrivado ? precioMostrar : ''}</span>
    </div>`;
  });

  if(elementos.length === 0) {
    listaHTML = '<i>Aún no hay equipos en el mapa.</i>';
  }

  document.getElementById('equiposLista').innerHTML = listaHTML;
  document.getElementById('totalPriceDisplay').textContent = modoPrivado ? `$ ${costoTotal}.00 USD` : '🔒 Oculto (Modo Cliente)';
}

function activarMedicion() {
  alert('Herramienta de medición activada: Toca dos puntos en el mapa.');
}

function exportarPropuesta() {
  alert(modoPrivado ? 'Generando lista de compras (BOM)...' : 'Generando presupuesto ciego para el cliente...');
}

resizeCanvas();
</script>
